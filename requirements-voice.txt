# V.O.I.D OPTIONAL voice dependencies (Phase 9A).
#
# Core V.O.I.D does NOT require these. Install only to use the voice interface:
#     .venv\Scripts\python.exe -m pip install -r requirements-voice.txt
#
# Windows SAPI TTS uses pywin32, which is already in requirements.txt.
# The Whisper model itself is downloaded on first use / explicit init - NOT
# during normal V.O.I.D startup (default model "small", ~460 MB, cached under
# %USERPROFILE%\.cache\huggingface). CPU int8 needs no GPU.

faster-whisper>=1.0     # local speech-to-text (bundles CTranslate2)
sounddevice>=0.4        # microphone capture
numpy>=1.24             # audio buffers
keyboard>=0.13          # global push-to-talk hotkey
openwakeword>=0.6       # local wake-word detection (Phase 9C; needs a custom
                        # "Hey V.O.I.D." model - no pretrained model is bundled)

# Optional: GPU speech-to-text (voice.stt_device: cuda). CTranslate2 bundles cuDNN but not cuBLAS.
# Measured on an RTX 5070: 1092 ms -> 115 ms mean decode at identical end-to-end accuracy. Without it
# V.O.I.D falls back to cpu/int8 automatically, so this is a speed-up, not a requirement.
nvidia-cublas-cu12; platform_system == "Windows"
